/* ============================================================
   TASK 2 - INBOUND 834 POLICY IDs MISSING FROM Enrollments_TEST
   PY2026
   READ-ONLY
   ============================================================ */

WITH InboundPolicies AS
(
    SELECT DISTINCT
        LTRIM(RTRIM(CAST(policy_id AS VARCHAR(100)))) AS policy_id
    FROM dbo.inbound_automation
    WHERE policy_id IS NOT NULL
      AND LTRIM(RTRIM(CAST(policy_id AS VARCHAR(100)))) <> ''
      AND coverage_year = 2026
),
EnrollmentPolicies AS
(
    SELECT DISTINCT
        LTRIM(RTRIM(CAST(policy_id AS VARCHAR(100)))) AS policy_id
    FROM dbo.Enrollments_TEST
    WHERE policy_id IS NOT NULL
      AND LTRIM(RTRIM(CAST(policy_id AS VARCHAR(100)))) <> ''
      AND coverage_year = 2026
)
SELECT
    i.policy_id
FROM InboundPolicies i
LEFT JOIN EnrollmentPolicies e
    ON i.policy_id = e.policy_id
WHERE e.policy_id IS NULL
ORDER BY i.policy_id;
